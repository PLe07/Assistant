#!/bin/zsh
# Convertit les fichiers .heic en .jpg (l'original est gardé).
# Proposé par le détecteur de corvées : lis-le avant de l'installer.
setopt null_glob 2>/dev/null || true  # aucun fichier : la boucle ne fait rien

dossier="$HOME"/'Downloads'
for f in "$dossier"/'IMG_'*'.heic'; do
  [ -e "$f" ] || continue
  sortie="${f%.*}.jpg"
